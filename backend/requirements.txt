# Hardened semantic cache — local embedding + vector store (nothing leaves the
# machine at runtime). sentence-transformers embeds prompts on-device (CPU-only)
# and ChromaDB is a local persistent vector store; both are installed/warmed at
# build time only and make no network call at request time. The all-MiniLM-L6-v2
# model artifact resolves from the local sentence-transformers / HF cache at
# runtime (fetched once at install/warm time, mirroring the en_core_web_sm note).
# Pinned to Python 3.13-compatible releases (cp313 wheels).
sentence-transformers==3.3.1
chromadb==1.5.9

# Hardware power tracking — on-device utilization sampling for the ESTIMATED tier
# (nothing leaves the machine at runtime). psutil reads CPU utilization locally
# (no network); estimated wattage = psutil.cpu_percent() × a configurable TDP
# model. On macOS psutil NEVER reports true wattage — it powers the ESTIMATED
# tier only. The MEASURED tier comes from the system `powermetrics` binary via a
# non-interactive probe gated behind POWER_TRY_MEASURED=1 (no Python dependency).
# Pinned to a Python 3.13-compatible (cp313) release.
psutil==6.1.0

# OpenTelemetry distributed tracing (feature: opentelemetry-tracing)
# Additive trace spans over the existing telemetry. Tracing defaults to a local,
# append-only JSONL file exporter (backend/logs/traces.jsonl) with ZERO outbound
# calls. The OTLP-over-HTTP exporter is used ONLY when TOKENQUICK_OTEL_EGRESS=true
# AND OTEL_EXPORTER_OTLP_ENDPOINT is set (the single Sanctioned_Egress) — it is
# lazily imported behind that gate, so the local-first zero-network default has
# no runtime dependency on the transport package. Pinned to 1.44.0 to match the
# version already present in the resolved tree (chromadb transitively depends on
# opentelemetry-*), avoiding a version conflict. cp313-compatible.
opentelemetry-api==1.44.0
opentelemetry-sdk==1.44.0
